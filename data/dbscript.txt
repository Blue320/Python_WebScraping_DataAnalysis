# Database 생성
mysql -u root –p
    # database 목록 확인
    show databases;
    # mysql DB를 사용
    use mysql;
    # python_db DB를 생성
    create database python_db;
    # DB 생성 후 목록 다시 확인하기
    show databases;
    # user는 python, password는 python 계정을 생성
    create user 'python'@'%' identified by 'python';
    # python_db에 대한 모든 권한을 python 계정에 부여
    grant all privileges on python_db.* to 'python'@'%';
    # 권한 부여 후 해당 내용을 DB에 즉시 적용
    flush privileges;
    # user 생성유무 확인
    select user, host from user;
    exit;

# python 계정의 password는 python
mysql -u python -p
    # 현재 python 계정의 권한을 확인
    show grants for current_user;
    use python_db;
    # table 목록 확인
    show tables;
    # songs DB 구조 확인
    desc songs;
    # songs DB 내에서 곡명을 출력
    select 곡명 from songs;
    # 특정 가수의 곡명을 추출하여 출력
    select 곡명 from songs where 가수 = '방탄소년단';
    select 곡명, 장르 from songs where 가수 = '방탄소년단';
